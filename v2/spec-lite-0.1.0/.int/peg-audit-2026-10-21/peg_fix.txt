%% guard: "a space after the first pipe" is text (Joseph, 2026-10-08). OPEN: which characters may follow | (the labels discussion); after a passed guard a bad name is an error, not text
attribute-start <- ":" &(NAME-FIRST / "\"" / "'") ;
%% guard: :name (rule:text-value-end), so " :-)" in text stays text. OPEN with the labels: NAME-FIRST
framed-semicolon <- ";" !"{" &(WS / EOL) ;
%% guard: a sameline comment needs a space each side (rule:framed-comments). OPEN: end of line as the right frame (def:framed); taken as yes here, and likewise for the framed backslash in :a \
fence-start <- FENCE !"{" ;
%% FENCE: three or more backticks with no later backtick on the line (rule:fence-opens-with-backticks, rule:fence-label-no-backtick). OPEN: TILDE fences (Joseph leans accept both)
reserved-start <- ("@" / "!") &(NAME-FIRST / ":" / "[" / "." / "<") ;
%% the rows of rule:reserved-spellings; @{ is a reference and !{ a reserved embed, both caught elsewhere
element <- element-start head _line-rest EOL _blocks EOB ;
%% the element's own line is value mode; after its EOL it holds blocks (prose mode). An element begun mid-line sits at its own column, so later lines nest under it (rule:sameline-element-is-child, rule:lines-follow-sameline-columns)
head <- name? key* trait* ;

%% OPEN: LABEL's characters; a quoted name is read by the delimited-string rules (rule:quoted-element-and-attribute-names)
key <- "[" (delimited-string / number / boolean / nil / angle / key-text) "]" ;
%% commit. rule:key-holds-a-value; inside, only \\ \[ \] escape (rule:key-escapes). OPEN: one key or several
key-text <- (embed / key-bare)+ ;

trait <- WS* "." LABEL ;
%% OPEN, provisional: spaces before a trait (rule:traits-may-follow-spaces)
_line-rest <- (_sep _item)* framed-comment? ;
_sep <- BS-SEP / WS ;
%% BS-SEP is the framed backslash. It only ends the text before it (R1 in the verbatim record, 2026-10-21) and is consumed; so in |a :href / \ Home, Home is a's $main
_item <- attribute / element / fence / ERROR-reserved / main ;
main <- value ;
%% a value with no attribute before it on the line is a value of the implicit attribute $main; several stack (rule:leftover-values)
attribute <- attribute-start name (WS (element / fence) / empty-text / WS value framed-comment? / ERROR-missing-value) ;
%% a sameline attribute takes one value (Joseph's lean, 2026-10-21), so in |el :c 1 z the z is $main. An element or fence only as that one value (rule:element-after-attribute-with-no-value, rule:fence-anywhere-structure). :a \ is an empty text (rule:backslash-empty-text)
empty-text <- &BS-SEP ;
ERROR-missing-value <- &(WS attribute-start / EOL) ;
%% rule:missing-value; :a :b is the reserved nested-attribute error (rule:attribute-after-attribute-with-no-value)
attribute-line <- attribute-start name (WS (element / fence) / empty-text / (_sep value framed-comment?)+ / ERROR-missing-value) EOL _values-below EOB ;
%% an attribute that starts its line: its values stack (rule:leftover-values, :x "one" two). OPEN: _values-below, the lines under an attribute (rule:attribute-bodies gap); Joseph, 2026-08-08: attributes are "a typed scalar until proven otherwise", so they are values, not prose
_values-below <- !"\u0000" ;
framed-comment <- WS framed-semicolon (WS RAW / &EOL) ;
%% attaches to the item before it (Joseph, 2026-10-20); on an element with no item before it, it is the element's next child. Deeper lines are its, with a warning (rule:framed-comment-owns-deeper-lines)
comment-block <- ";" !"{" RAW EOB ;

fence <- fence-start head _line-rest EOL FENCE-BODY fence-closer ;
%% the opener's line reads like an element's line (rule:fence-opener-line); the default handler accepts keys, the trait exact and a comment, post-lexically (rule:fence-default-handler). FENCE-BODY: raw lines to the closer, each keeping its ending, dedented by the handler unless exact
fence-closer <- FENCE-CLOSE framed-comment? EOL ;
%% FENCE-CLOSE: at least the opener's count of backticks, first on its line (rule:fence-closes-at-n-or-more); anything else after them is an error (rule:fence-closer-trailing-text-is-error). ERROR-reserved <- reserved-start ;
